/** * ProjectFlow — Slack Reminder Hook * File: pb_hooks/slack_reminders.pb.js * * Drop this file into your PocketBase `pb_hooks/` directory. * PocketBase will pick it up automatically on next restart (no recompile needed). * * What it does: * - Runs every 5 minutes (server-side cron) * - Reads your ProjectFlow state from the `projectflow` collection * - Finds tasks with a due Slack reminder that haven't been sent yet * - POSTs a formatted message to your Slack incoming webhook * - Marks each reminder as sent so it doesn't fire again * - Saves the updated state back to PocketBase * * Requirements: * - PocketBase v0.16+ (uses cronAdd and $http) * - Your ProjectFlow data must be stored in a collection called `projectflow` * with a `data` JSON field (the default setup used by the app) * - A Slack Incoming Webhook URL stored in S.slackWebhook inside that data * * Setup: * 1. Copy this file to /pb_hooks/slack_reminders.pb.js * 2. Restart PocketBase (or it may hot-reload automatically) * 3. Check PocketBase logs to confirm the cron registered */ // --------------------------------------------------------------------------- // Helper: format a date string for display // --------------------------------------------------------------------------- function fmtDate(iso) { if (!iso) return ''; try { const d = new Date(iso); return d.toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }); } catch (e) { return iso; } } // --------------------------------------------------------------------------- // Cron: runs every 5 minutes // --------------------------------------------------------------------------- cronAdd('slack-reminders', '*/5 * * * *', function () { // ── 1. Load the ProjectFlow state record ────────────────────────────────── let record; try { // The app stores everything in a single record. Adjust the filter if your // collection uses a different key or has multiple rows per user. record = $app.dao().findFirstRecordByFilter('projectflow', '1=1'); } catch (e) { // Collection doesn't exist yet or is empty — nothing to do return; } let S; try { S = JSON.parse(record.getString('data') || '{}'); } catch (e) { console.error('[slack-reminders] Could not parse projectflow data:', e); return; } const webhook = S.slackWebhook; if (!webhook) return; // No webhook configured — skip const tasks = S.tasks || []; const projects = S.projects || []; const members = S.members || []; const now = new Date(); let dirty = false; // ── 2. Find tasks with due, unsent reminders ────────────────────────────── for (const t of tasks) { const r = t.slackReminder; if (!r || !r.datetime || r.sent) continue; if (new Date(r.datetime) > now) continue; // not due yet // ── 3. Build Slack Block Kit message ──────────────────────────────────── const proj = projects.find(p => p.id === t.projectId); const assigneeNames = (t.assignees || []) .map(id => { const m = members.find(x => x.id === id); return m ? m.name : null; }) .filter(Boolean) .join(', '); const projStr = proj ? `\n*Project:* ${proj.name}` : ''; const phaseStr = t.phase ? `\n*Phase:* ${t.phase}` : ''; const dueStr = t.endDate ? `\n*Due:* ${fmtDate(t.endDate)}` : ''; const assignStr = assigneeNames ? `\n*Assigned to:* ${assigneeNames}` : ''; const noteStr = r.message ? `\n\n📝 _${r.message}_` : ''; const doneStr = t.done ? ' · ✅ Completed' : ''; const prioEmoji = { Critical: '🔴', High: '🟠', Medium: '🟡', Low: '🟢' }[t.priority] || '⚪'; const nameStr = t.done ? `~*${t.name}*~` : `*${t.name}*`; const payload = { text: `⏰ Task reminder: ${t.name}`, blocks: [ { type: 'header', text: { type: 'plain_text', text: '⏰ Task Reminder', emoji: true } }, { type: 'section', text: { type: 'mrkdwn', text: `${nameStr}${projStr}${phaseStr}${dueStr}${assignStr}\n${prioEmoji} *${t.priority || 'Medium'} priority*${doneStr}${noteStr}` } }, { type: 'context', elements: [ { type: 'mrkdwn', text: `Sent from *Airwiggles ProjectFlow* · ${now.toISOString().replace('T', ' ').slice(0, 16)} UTC` } ] } ] }; // ── 4. POST to Slack ───────────────────────────────────────────────────── try { const res = $http.send({ url: webhook, method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload), timeout: 10, // seconds }); if (res.statusCode >= 200 && res.statusCode < 300) { t.slackReminder.sent = true; dirty = true; console.log(`[slack-reminders] Sent reminder for task: ${t.name}`); } else { console.error(`[slack-reminders] Slack returned ${res.statusCode} for task: ${t.name}`); } } catch (e) { console.error(`[slack-reminders] HTTP error for task ${t.name}:`, e); } } // ── 5. Save state back if anything changed ──────────────────────────────── if (dirty) { try { record.set('data', JSON.stringify(S)); $app.dao().saveRecord(record); console.log('[slack-reminders] State saved after sending reminders.'); } catch (e) { console.error('[slack-reminders] Failed to save updated state:', e); } } });
